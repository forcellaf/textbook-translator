% CUSTOM COMMANDS
% ---------------------------------------------------------
% Standard figure. Every figure in the book goes through this, so they are
% all centred, all the same width, and all captioned the same way.
%
%   \bookfig{IMG_0042}{Charge distribution inside a proton}
%   \bookfig[0.85]{IMG_0107}{A wider diagram}
%
\newcommand{\bookfig}[3][0.6]{%
  \begin{figure}[htbp]
    \centering
    \includegraphics[width=#1\textwidth,keepaspectratio]{#2}
    \caption{#3}
  \end{figure}%
}

% Two figures side by side, e.g. a source diagram split into (a) and (b).
%
%   \bookfigtwo{IMG_0001}{(a)}{IMG_0002}{(b)}{Overall caption}
%
\newcommand{\bookfigtwo}[5]{%
  \begin{figure}[htbp]
    \centering
    \begin{minipage}{0.45\textwidth}
      \centering\includegraphics[width=\textwidth,keepaspectratio]{#1}\\[2pt]{\small #2}
    \end{minipage}\hfill
    \begin{minipage}{0.45\textwidth}
      \centering\includegraphics[width=\textwidth,keepaspectratio]{#3}\\[2pt]{\small #4}
    \end{minipage}
    \caption{#5}
  \end{figure}%
}
